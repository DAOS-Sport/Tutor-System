### 1-A. 建立 employees 資料表

```sql
CREATE TABLE employees (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  ragic_employee_id   VARCHAR(50) UNIQUE,
  employee_number     VARCHAR(50) UNIQUE,
  name                VARCHAR(100) NOT NULL,
  phone               VARCHAR(20) UNIQUE,
  email               VARCHAR(255) UNIQUE,
  line_uid            VARCHAR(100) UNIQUE,
  password_hash       VARCHAR(255),
  roles               TEXT[] NOT NULL DEFAULT '{}',
  venue_id            VARCHAR(10) REFERENCES venues(id),
  is_senior           BOOLEAN NOT NULL DEFAULT FALSE,
  pricing_multiplier  DECIMAL(5,2) NOT NULL DEFAULT 1.00,
  bio_rich_text       TEXT,
  specialties         TEXT[],
  is_active           BOOLEAN NOT NULL DEFAULT TRUE,
  last_synced_at      TIMESTAMPTZ,
  last_login_at       TIMESTAMPTZ,
  created_at          TIMESTAMPTZ DEFAULT NOW(),
  updated_at          TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_employees_phone    ON employees(phone);
CREATE INDEX idx_employees_email    ON employees(email);
CREATE INDEX idx_employees_line_uid ON employees(line_uid);
CREATE INDEX idx_employees_roles    ON employees USING GIN(roles);
```

### 1-B. 從 coaches 遷移（roles = ['coach']）

```sql
INSERT INTO employees (
  ragic_employee_id, employee_number, name, phone, line_uid, email,
```

### 1-C. 從 admin_users 遷移（role 字串 → roles 陣列）

admin_users.role 對應關係：
- `'admin'` → `'system_admin'`
- `'manager'` → `'manager'`
- `'staff'` → `'counter'`

```sql
-- 已有的員工（從 coaches 遷移過來的），補充後台欄位
UPDATE employees e
SET
  email = COALESCE(e.email, au.username),
  password_hash = COALESCE(e.password_hash, au.password_hash),
  venue_id = COALESCE(e.venue_id, au.venue_id),
  roles = (
    SELECT array_agg(DISTINCT r)
    FROM unnest(array_append(e.roles,
      CASE au.role
        WHEN 'admin'   THEN 'system_admin'
        WHEN 'manager' THEN 'manager'
        WHEN 'staff'   THEN 'counter'
        ELSE au.role
      END
    )) AS r
  )
FROM admin_users au
JOIN admin_staff ast ON ast.id = au.id
WHERE e.phone = ast.phone;

-- 純後台人員（不在 coaches 中），建立新記錄
INSERT INTO employees (name, email, password_hash, roles, venue_id, is_active)
SELECT
  au.name,
  au.username,
  au.password_hash,
  ARRAY[CASE au.role WHEN 'admin' THEN 'system_admin' WHEN 'manager' THEN 'manager' ELSE 'counter' END]::TEXT[],
  au.venue_id,
  TRUE
FROM admin_users au
WHERE NOT EXISTS (
  SELECT 1 FROM employees e WHERE e.email = au.username
)
ON CONFLICT (email) DO NOTHING;
```

### 1-D. 補充 ragic_employee_id（從 admin_staff）
